from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy
from sqlalchemy import text

app = Flask(__name__)
# Utiliza SQLite em memória para ambiente de testes rápidos
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///:memory:'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False

db = SQLAlchemy(app)

# Modelo de banco de dados
class User(db.Model):
    __tablename__ = 'users'
    id = db.Column(db.Integer, primary_key=True)
    email = db.Column(db.String(120), unique=True, nullable=False)
    password = db.Column(db.String(120), nullable=False)

# Inicializa o banco de dados e adiciona um usuário de teste
with app.app_context():
    db.create_all()
    user_admin = User(email='admin@empresa.com', password='SenhaSuperSecreta123')
    db.session.add(user_admin)
    db.session.commit()

@app.route('/api/login', methods=['POST'])
def login():
    data = request.get_json() or {}
    email = data.get('email', '')
    password = data.get('password', '')

    # VULNERABILIDADE: Interpolação direta de strings em consulta SQL bruta
    query = f"SELECT * FROM users WHERE email = '{email}' AND password = '{password}'"
    
    try:
        user = db.session.execute(text(query)).fetchone()
        if user:
            # Retorna o email do usuário autenticado se a consulta retornar algum registro
            return jsonify({"status": "success", "user": user.email})
        return jsonify({"status": "error", "message": "Credenciais inválidas"}), 401
    except Exception as e:
        return jsonify({"status": "error", "message": str(e)}), 500

if __name__ == '__main__':
    app.run(host='127.0.0.1', port=5000, debug=True)
